2. ALKALMAZÁS: potpotpotzh-vote-feladat (Szavazó felület)
Itt nincsenek biztonsági korlátozások (nincs login), a felület aszinkron AJAX hívásokkal (REST API) és egyedi tag library-vel működik.

src/main/webapp/WEB-INF/tags/best.tag (A legjobb előadást kiíró egyedi Taglib)

Java
<%@ tag pageEncoding="UTF-8" %>
<div style="margin: 20px 0; padding: 15px; background-color: #e8f5e9; border: 2px solid #4caf50; border-radius: 5px; max-width: 800px;">
    <h3 style="margin-top: 0; color: #2e7d32;">A te legjobb előadásod:</h3>
    <% 
        String bestPresentation = "Még nem jelölt legjobb előadást.";
        jakarta.servlet.http.Cookie[] cookies = request.getCookies();
        if(cookies != null) {
            for(jakarta.servlet.http.Cookie c : cookies) {
                if("bestPresentation".equals(c.getName())) {
                    // Dekódoljuk a JS encodeURIComponent által készített stringet
                    bestPresentation = java.net.URLDecoder.decode(c.getValue(), java.nio.charset.StandardCharsets.UTF_8);
                }
            }
        }
    %>
    <p style="font-size: 1.2em; font-weight: bold; margin-bottom: 0;">
        <%= bestPresentation %>
    </p>
</div>
src/main/webapp/index.jsp (Fő szavazó felület)

Java
<%@ page contentType="text/html;charset=UTF-8" language="java" pageEncoding="UTF-8" %>
<%@ taglib uri="jakarta.tags.core" prefix="c" %>
<%@ taglib prefix="mytags" tagdir="/WEB-INF/tags" %>
<!DOCTYPE html>
<html lang="hu">
<head>
    <title>Konferencia Értékelő Rendszer</title>
    <style>
        body { font-family: Arial, sans-serif; margin: 30px; }
        table { border-collapse: collapse; width: 100%; max-width: 1000px; }
        th, td { border: 1px solid #aaa; padding: 10px; text-align: left; }
        th { background-color: #f2f2f2; }
        .rating-btns button { margin-right: 5px; cursor: pointer; padding: 5px 10px; }
        .best-btn { background-color: #ff9800; color: white; border: none; padding: 5px 10px; cursor: pointer; }
    </style>
    <script>
        // 11. Feladat: AJAX hívás a REST endpoint felé (Jersey)
        function vote(id, rating) {
            fetch('${pageContext.request.contextPath}/api/presentations/' + id + '/vote', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/x-www-form-urlencoded'
                },
                body: 'rating=' + rating
            })
            .then(response => {
                if(response.ok) {
                    location.reload(); // Frissítjük az oldalt az új átlagért
                } else {
                    alert('Hiba történt a szavazás során!');
                }
            });
        }

        // 12. Feladat: Süti (Cookie) beállítása a legjobb előadásra
        function setBest(title, speaker) {
            let encodedVal = encodeURIComponent(title + " - " + speaker);
            // Süti beállítása 1 napos lejárattal
            document.cookie = "bestPresentation=" + encodedVal + "; path=/; max-age=86400";
            location.reload(); // Frissítünk, hogy a Taglib egyből kiírja az eredményt
        }
    </script>
</head>
<body>
    <h2>Értékeld a Konferencia Előadásait!</h2>
    
    <mytags:best />

    <table>
        <tr>
            <th>Cím</th>
            <th>Előadó</th>
            <th>Téma</th>
            <th>Átlagértékelés</th>
            <th>Szavazás (1-5)</th>
            <th>Kiemelés</th>
        </tr>
        <%-- A VoteListServlet hívja be a webservice-t és adja át a 'list' attribútumot --%>
        <c:forEach var="p" items="${list}">
            <tr>
                <td><strong>${p.title}</strong></td>
                <td>${p.speakerName}</td>
                <td>${p.topic}</td>
                <td>
                    <c:choose>
                        <c:when test="${p.ratingCount > 0}">
                            ${Math.round((p.ratingSum * 10.0) / p.ratingCount) / 10.0} 
                            <span style="font-size: 0.8em; color: #666;">(${p.ratingCount} db)</span>
                        </c:when>
                        <c:otherwise>Nincs adat</c:otherwise>
                    </c:choose>
                </td>
                <td class="rating-btns">
                    <button onclick="vote(${p.id}, 1)">1</button>
                    <button onclick="vote(${p.id}, 2)">2</button>
                    <button onclick="vote(${p.id}, 3)">3</button>
                    <button onclick="vote(${p.id}, 4)">4</button>
                    <button onclick="vote(${p.id}, 5)">5</button>
                </td>
                <td>
                    <button class="best-btn" onclick="setBest('${p.title}', '${p.speakerName}')">
                        ★ Legjobb előadás
                    </button>
                </td>
            </tr>
        </c:forEach>
    </table>
</body>
</html>
