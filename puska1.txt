5. Lépés: A HallgatoRepository befejezése
A feladat megköveteli, hogy lehessen listázni, azonosító alapján keresni, létrehozni és törölni. Egészítsd ki a HallgatoRepository osztályt a következő metódusokkal. A try-with-resources blokkok gondoskodnak a kapcsolatok és a statementek megfelelő lezárásáról.  
PDF
+ 1

Java
package hu.pte.mik.prog4.potzh_2026.repository;

import hu.pte.mik.prog4.potzh_2026.entity.Hallgato;

import javax.naming.Context;
import javax.naming.InitialContext;
import javax.sql.DataSource;
import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.util.ArrayList;
import java.util.List;

public class HallgatoRepository {

    private Connection getConnection() throws Exception {
        Context initContext = new InitialContext();
        Context envContext = (Context) initContext.lookup("java:/comp/env");
        DataSource ds = (DataSource) envContext.lookup("jdbc/PotPotZH_university_MariaDB");
        return ds.getConnection();
    }

    public List<Hallgato> findAll() {
        List<Hallgato> hallgatok = new ArrayList<>();
        String sql = "SELECT * FROM hallgato ORDER BY hallgato_nev ASC";
        
        try (Connection conn = getConnection();
             PreparedStatement ps = conn.prepareStatement(sql);
             ResultSet rs = ps.executeQuery()) {
             
            while (rs.next()) {
                hallgatok.add(new Hallgato(
                        rs.getInt("ID"),
                        rs.getString("hallgato_nev"),
                        rs.getString("szak"),
                        rs.getInt("felev"),
                        rs.getDouble("egyetemi_atlag")
                ));
            }
        } catch (Exception e) {
            e.printStackTrace();
        }
        return hallgatok;
    }

    public Hallgato findById(int id) {
        String sql = "SELECT * FROM hallgato WHERE ID = ?";
        try (Connection conn = getConnection();
             PreparedStatement ps = conn.prepareStatement(sql)) {
             
            ps.setInt(1, id);
            try (ResultSet rs = ps.executeQuery()) {
                if (rs.next()) {
                    return new Hallgato(
                            rs.getInt("ID"),
                            rs.getString("hallgato_nev"),
                            rs.getString("szak"),
                            rs.getInt("felev"),
                            rs.getDouble("egyetemi_atlag")
                    );
                }
            }
        } catch (Exception e) {
            e.printStackTrace();
        }
        return null;
    }

    public void save(Hallgato hallgato) {
        String sql = "INSERT INTO hallgato (hallgato_nev, szak, felev, egyetemi_atlag) VALUES (?, ?, ?, ?)";
        try (Connection conn = getConnection();
             PreparedStatement ps = conn.prepareStatement(sql)) {
             
            ps.setString(1, hallgato.getHallgatoNev());
            ps.setString(2, hallgato.getSzak());
            ps.setInt(3, hallgato.getFelev());
            ps.setDouble(4, hallgato.getEgyetemiAtlag());
            ps.executeUpdate();
        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    public void delete(int id) {
        String sql = "DELETE FROM hallgato WHERE ID = ?";
        try (Connection conn = getConnection();
             PreparedStatement ps = conn.prepareStatement(sql)) {
             
            ps.setInt(1, id);
            ps.executeUpdate();
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}

A leírás szerint a service réteg minden repository metódushoz kell, hogy tartalmazzon egy hívást. Hozd létre a hu.pte.mik.prog4.potzh_2026.service csomagot, benne a HallgatoService osztállyal.

package hu.pte.mik.prog4.potzh_2026.service;

import hu.pte.mik.prog4.potzh_2026.entity.Hallgato;
import hu.pte.mik.prog4.potzh_2026.repository.HallgatoRepository;

import java.util.List;

public class HallgatoService {

    private final HallgatoRepository repository;

    public HallgatoService() {
        this.repository = new HallgatoRepository();
    }

    public List<Hallgato> getAllHallgato() {
        return repository.findAll();
    }

    public Hallgato getHallgatoById(int id) {
        return repository.findById(id);
    }

    public void saveHallgato(Hallgato hallgato) {
        repository.save(hallgato);
    }

    public void deleteHallgato(int id) {
        repository.delete(id);
    }
}

package hu.pte.mik.prog4.potzh_2026.ws;

import hu.pte.mik.prog4.potzh_2026.entity.Hallgato;
import hu.pte.mik.prog4.potzh_2026.service.HallgatoService;

import jakarta.jws.WebMethod;
import jakarta.jws.WebParam;
import jakarta.jws.WebService;
import java.util.List;

@WebService(serviceName = "HallgatoDataService")
public class SoapHallgatoService {

    private final HallgatoService service = new HallgatoService();

    @WebMethod
    public List<Hallgato> getHallgatok() {
        return service.getAllHallgato();
    }

    @WebMethod
    public Hallgato getHallgato(@WebParam(name = "id") int id) {
        return service.getHallgatoById(id);
    }

    @WebMethod
    public void createHallgato(
            @WebParam(name = "nev") String nev,
            @WebParam(name = "szak") String szak,
            @WebParam(name = "felev") int felev,
            @WebParam(name = "atlag") double atlag) {
            
        Hallgato h = new Hallgato(0, nev, szak, felev, atlag);
        service.saveHallgato(h);
    }

    @WebMethod
    public void deleteHallgato(@WebParam(name = "id") int id) {
        service.deleteHallgato(id);
    }
}

        <servlet-class>com.sun.xml.ws.transport.http.servlet.WSServlet</servlet-class>

Java
package hu.pte.mik.prog4.potzh_2026.api.controller;

import hu.pte.mik.prog4.potzh_2026.entity.Hallgato;
import hu.pte.mik.prog4.potzh_2026.service.HallgatoService;

import jakarta.servlet.ServletException;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.HttpServlet;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import java.io.IOException;
import java.io.PrintWriter;

@WebServlet("/api/hallgato")
public class HallgatoApiController extends HttpServlet {

    private final HallgatoService service = new HallgatoService();

    // GET /api/hallgato?id=1 -> Visszaadja az 1-es ID-jú hallgatót (egyszerű szövegként)
    @Override
    protected void doGet(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
        resp.setContentType("text/plain;charset=UTF-8");
        String idParam = req.getParameter("id");
        
        try (PrintWriter out = resp.getWriter()) {
            if (idParam != null) {
                Hallgato h = service.getHallgatoById(Integer.parseInt(idParam));
                if (h != null) {
                    out.println("ID: " + h.getId() + ", Név: " + h.getHallgatoNev() + ", Szak: " + h.getSzak());
                } else {
                    out.println("Nem található hallgató ezzel az ID-val.");
                }
            } else {
                out.println("Kérlek adj meg egy 'id' paramétert!");
            }
        }
    }

    // POST /api/hallgato -> Új hallgató mentése (paraméterekből)
    @Override
    protected void doPost(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
        String nev = req.getParameter("nev");

3. Lépés: Az AuthenticationModule (LoginModule)
Ez a lelke a JAAS-nek. Itt csatlakozunk a security adatbázishoz, ellenőrizzük a BCrypt jelszót, és betöltjük a szerepköröket.

AuthenticationModule.java

Java
package hu.pte.mik.prog4.potzh_2026.jaas;

import org.mindrot.jbcrypt.BCrypt;

import javax.naming.Context;
import javax.naming.InitialContext;
import javax.security.auth.Subject;
import javax.security.auth.callback.*;
import javax.security.auth.login.LoginException;
import javax.security.auth.spi.LoginModule;
import javax.sql.DataSource;
import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.util.ArrayList;
import java.util.List;
import java.util.Map;

public class AuthenticationModule implements LoginModule {

    private Subject subject;
    private CallbackHandler callbackHandler;
    private String username;
    private boolean loginSucceeded = false;
    private List<String> roles = new ArrayList<>();

    @Override
    public void initialize(Subject subject, CallbackHandler callbackHandler, Map<String, ?> sharedState, Map<String, ?> options) {
        this.subject = subject;
        this.callbackHandler = callbackHandler;
    }

    @Override
    public boolean login() throws LoginException {
        if (callbackHandler == null) throw new LoginException("No CallbackHandler defined");

        NameCallback nameCallback = new NameCallback("Username:");
        PasswordCallback passwordCallback = new PasswordCallback("Password:", false);

        try {
            callbackHandler.handle(new Callback[]{nameCallback, passwordCallback});
            username = nameCallback.getName();
            String password = new String(passwordCallback.getPassword());

            Context initContext = new InitialContext();
            Context envContext = (Context) initContext.lookup("java:/comp/env");
            DataSource ds = (DataSource) envContext.lookup("jdbc/PotPotZH_security_MariaDB");

            try (Connection conn = ds.getConnection()) {
                // 1. Felhasználó jelszavának lekérdezése
                String userSql = "SELECT id, jelszo FROM felhasznalo WHERE felhasznalonev = ?";
                try (PreparedStatement ps = conn.prepareStatement(userSql)) {
                    ps.setString(1, username);
                    try (ResultSet rs = ps.executeQuery()) {
                        if (rs.next()) {
                            String hashedPassword = rs.getString("jelszo");
                            int userId = rs.getInt("id");
                            
                            // BCrypt jelszó ellenőrzés
                            if (BCrypt.checkpw(password, hashedPassword)) {
                                loginSucceeded = true;
                                
                                // 2. Szerepkörök betöltése
                                String roleSql = "SELECT s.kod FROM szerepkor s " +
                                                 "JOIN felhasznalo_szerepkor fs ON s.id = fs.szerepkor_id " +
                                                 "WHERE fs.felhasznalo_id = ?";
                                try (PreparedStatement psRole = conn.prepareStatement(roleSql)) {
                                    psRole.setInt(1, userId);
                                    try (ResultSet rsRole = psRole.executeQuery()) {
                                        while (rsRole.next()) {
                                            roles.add(rsRole.getString("kod"));
                                        }
                                    }
                                }
                            }
                        }
                    }
                }
            }
        } catch (Exception e) {
            throw new LoginException("Adatbázis hiba az authentikáció során: " + e.getMessage());
        }

        if (!loginSucceeded) {
            throw new LoginException("Hibás felhasználónév vagy jelszó!");
        }

        return true;
    }

    @Override
    public boolean commit() throws LoginException {
        if (!loginSucceeded) return false;

        subject.getPrincipals().add(new UserPrincipal(username));
        for (String role : roles) {
            subject.getPrincipals().add(new RolePrincipal(role));
        }
        return true;
    }

    @Override
    public boolean abort() throws LoginException {
        return false;
    }

    @Override
    public boolean logout() throws LoginException {
        subject.getPrincipals().clear();
        return true;
    }
}

            if (h != null) {
                // Konzolra logolás XML formátumban [cite: 71]
                System.out.println("<?xml version=\"1.0\" encoding=\"UTF-8\"?>");
                System.out.println("<hallgato>");
                System.out.println("  <id>" + h.getId() + "</id>");
                System.out.println("  <nev>" + h.getHallgatoNev() + "</nev>");
                System.out.println("  <szak>" + h.getSzak() + "</szak>");
                System.out.println("  <felev>" + h.getFelev() + "</felev>");
                System.out.println("  <atlag>" + h.getEgyetemiAtlag() + "</atlag>");
                System.out.println("</hallgato>");
            }
        }
        resp.sendRedirect("search-servlet");
    }
}

package hu.pte.mik.prog4.potzh_2026.servlet;

import hu.pte.mik.prog4.potzh_2026.service.HallgatoKliensService;

import jakarta.servlet.ServletException;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.HttpServlet;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import java.io.IOException;

@WebServlet("/save-servlet")
public class SaveServlet extends HttpServlet {

    private final HallgatoKliensService service = new HallgatoKliensService();

    @Override
    protected void doPost(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
        req.setCharacterEncoding("UTF-8");
        String nev = req.getParameter("nev");
        String szak = req.getParameter("szak");
        int felev = Integer.parseInt(req.getParameter("felev"));
        double atlag = Double.parseDouble(req.getParameter("atlag"));

        service.save(nev, szak, felev, atlag);
        resp.sendRedirect("search-servlet");
    }
}
